# ALIGNMENT -- CGB palettes ("high contrast black": black void, neon purple,
# neon green, white highlights).
#
#   @bgpal   <n> <NAME> <c0> <c1> <c2> <c3>
#   @sprpal  <n> <NAME> <c0> <c1> <c2> <c3>
#   @deckpal <n> <NAME> <c0> <c1> <c2> <c3>   hull variants swapped into PAL_WALL
#   @alias   <NAME> <TARGET>                   extra #define in assets.h
#
# Colours are 24-bit #rrggbb; the generator converts to RGB555 (>>3).
# c0 = black, c1 = dim, c2 = bright, c3 = hot highlight.
# Every BG palette MUST have c0 = #000000.  Sprite c0 is transparent.
